CMAKE_MINIMUM_REQUIRED(VERSION 3.5)
PROJECT(RVO VERSION 1.0.0)

#COMPILE SETTINGS
SET(CMAKE_BUILD_TYPE Release CACHE STRING "Compiler optimization level")
SET_PROPERTY(CACHE CMAKE_BUILD_TYPE PROPERTY STRINGS Debug Release)
IF(NOT CMAKE_BUILD_TYPE)
  SET(CMAKE_BUILD_TYPE Release)
ENDIF()
SET(CMAKE_MODULE_PATH ${CMAKE_MODULE_PATH} ${PROJECT_SOURCE_DIR}/cmake/ ${PROJECT_SOURCE_DIR}/extern/eigen/cmake)
IF(CMAKE_BUILD_TYPE STREQUAL "Debug")
  MESSAGE(STATUS "Build type: Debug")
  IF(MSVC)
    SET(CMAKE_CXX_FLAGS "/bigobj -W0 ${CMAKE_CXX_FLAGS}")
  ELSE(MSVC)
    ADD_COMPILE_DEFINITIONS(_GLIBCXX_DEBUG=1)
    SET(CMAKE_CXX_FLAGS "-O0 -Wall -fPIC -std=c++11 -Wno-unused-function -Wno-unused-local-typedefs ${CMAKE_CXX_FLAGS}")
  ENDIF(MSVC)
ELSE()
  MESSAGE(STATUS "Build type: Release")
  IF(MSVC)
    SET(CMAKE_CXX_FLAGS "/bigobj -W0 ${CMAKE_CXX_FLAGS}")
  ELSE(MSVC)
    SET(CMAKE_CXX_FLAGS "-O3 -Wall -fPIC -std=c++11 -Wno-unused-function -Wno-unused-local-typedefs ${CMAKE_CXX_FLAGS}")
  ENDIF(MSVC)
ENDIF()
SET(CMAKE_CONFIGURATION_TYPES ${CMAKE_BUILD_TYPE})

#C++11 ABI
OPTION(USE_CXX11_ABI "Whether to use C++11's ABI" ON)
IF(NOT ${USE_CXX11_ABI})
  MESSAGE(STATUS "Disabling C++11's ABI")
  SET(CMAKE_CXX_FLAGS "-D_GLIBCXX_USE_CXX11_ABI=0 ${CMAKE_CXX_FLAGS}")
ELSE()
  MESSAGE(STATUS "Enabling C++11's ABI")
ENDIF()

#LIBRARY SETTINGS
SET(CGAL_DO_NOT_WARN_ABOUT_CMAKE_BUILD_TYPE TRUE)
INCLUDE("${PROJECT_SOURCE_DIR}/cmake/LoadLibs.cmake")
ADD_DEFINITIONS(-DDATA_PATH="${PROJECT_SOURCE_DIR}/data")
#let user choose the floating point precision
SET(FLOAT_PRECISION double CACHE STRING "Multi-precision setting")
IF(QUADMATH_FOUND AND MPFR_FOUND)
  SET_PROPERTY(CACHE FLOAT_PRECISION PROPERTY STRINGS double float128 mpfr_float)
ELSEIF(QUADMATH_FOUND)
  SET_PROPERTY(CACHE FLOAT_PRECISION PROPERTY STRINGS double float128)
ELSEIF(MPFR_FOUND)
  SET_PROPERTY(CACHE FLOAT_PRECISION PROPERTY STRINGS double mpfr_float)
ELSE()
  SET(FLOAT_PRECISION double)
ENDIF()
ADD_DEFINITIONS(-DLSCALAR=${FLOAT_PRECISION})
IF(${FLOAT_PRECISION} STREQUAL "double")
ELSE()
  ADD_DEFINITIONS(-DFORCE_ADD_DOUBLE_PRECISION)
ENDIF()

#THIRD PARTY HEADERS
INCLUDE_DIRECTORIES(${PROJECT_SOURCE_DIR})
MACRO(SEARCH_SOURCE NAME DIRPRJ)
  FILE(GLOB_RECURSE header${NAME} ${PROJECT_SOURCE_DIR}/${DIRPRJ}/*.h ${PROJECT_SOURCE_DIR}/${DIRPRJ}/*.hpp ${PROJECT_SOURCE_DIR}/${DIRPRJ}/*.hh)
  FILE(GLOB_RECURSE source${NAME} ${PROJECT_SOURCE_DIR}/${DIRPRJ}/*.cpp ${PROJECT_SOURCE_DIR}/${DIRPRJ}/*.cc ${PROJECT_SOURCE_DIR}/${DIRPRJ}/*.c)
ENDMACRO(SEARCH_SOURCE)
SEARCH_SOURCE(RVO RVO)
LIST(REMOVE_ITEM sourceRVO ${PROJECT_SOURCE_DIR}/RVO/Python.cpp)

#BUILD LIBRARY
ADD_LIBRARY(RVO SHARED ${sourceRVO} ${headerRVO})
TARGET_LINK_LIBRARIES(RVO stdc++fs ${ALL_LIBRARIES} ${TinyVisualizer_LIBRARIES})
SET_TARGET_PROPERTIES(RVO PROPERTIES PUBLIC_HEADER "${headerRVO}")

#INSTALL LIBRARY
INSTALL(TARGETS RVO
  LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/${CMAKE_BUILD_TYPE}
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}/${CMAKE_BUILD_TYPE}
  PUBLIC_HEADER DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/RVO)

#Remove visualizer
MACRO(REMOVE_VISUALIZER SRC_FILES)
  STRING(CONCAT TAR_FILES ${SRC_FILES} "NoVis")
  FOREACH(ITEM ${${SRC_FILES}})
    STRING(FIND "${ITEM}" "Visualizer" RES)
    IF(${RES} LESS 0)
      LIST(APPEND ${TAR_FILES} "${ITEM}")
    ENDIF()
  ENDFOREACH()
ENDMACRO()

#Pybind11-Python
IF(${FLOAT_PRECISION} STREQUAL "double")    #Python interface is supported by the double version
  IF(NOT (PYTHON_BINDING STREQUAL "None"))
    #We do not export visualizer functions
    REMOVE_VISUALIZER(sourceRVO)
    REMOVE_VISUALIZER(headerRVO)
    #Create pybind11 library
    ADD_SUBDIRECTORY(${PROJECT_SOURCE_DIR}/extern/pybind11)
    PYBIND11_ADD_MODULE(pyRVO SHARED ${PROJECT_SOURCE_DIR}/RVO/Python.cpp ${sourceRVONoVis} ${headerRVONoVis})
    TARGET_LINK_LIBRARIES(pyRVO PRIVATE stdc++fs ${ALL_LIBRARIES})
    #INSTALL python
    INSTALL(TARGETS pyRVO 
      LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
      ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
      PUBLIC_HEADER DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
  ENDIF()
ENDIF()

#EXE
MACRO(ADD_EXE NAME)
  ADD_EXECUTABLE(${NAME} Main/${NAME}.cpp)
  TARGET_LINK_LIBRARIES(${NAME} RVO ${ALL_STATIC_LIBRARIES})
ENDMACRO(ADD_EXE)
ADD_EXE(mainDebugRVO)
ADD_EXE(mainSimulator)
ADD_EXE(mainSimulatorSource)
ADD_EXE(mainMultiSimulator)
ADD_EXE(mainMultiSimulatorSource)
ADD_EXE(mainMultiVisibility)
ADD_EXE(mainStreetSimulator)
ADD_EXE(mainVisibility)
